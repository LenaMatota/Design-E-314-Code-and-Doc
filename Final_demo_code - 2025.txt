uint8_t screen_history[MAX_HISTORY];
int history_index = 0;

void push_screen(uint8_t screen) {
    if (history_index < MAX_HISTORY) {
        screen_history[history_index++] = screen;
    }
}

uint8_t pop_screen(void) {
    if (history_index > 0) {
        return screen_history[--history_index];
    }
    return 0; // fallback default screen
}

void Process_ADC_Data(void) {
    float Voltage_Array[BUFFER_SIZE / 2];
    float Current_Array[BUFFER_SIZE / 2];
    float max_voltage = -1000.0f;
    float max_current = -1000.0f;
    int max_voltage_index = 0;
    int max_current_index = 0;
    float voltage_zero_time = 0.0f;
    float current_zero_time = 0.0f;
    const float ZERO_THRESHOLD = 0.03f; // Increased to 0.03V from 0.02V
    const float CROSSING_THRESHOLD = 0.01f; // Tolerance for zero-crossing detection
    float voltage_zero_sum = 0.0f;
    float current_zero_sum = 0.0f;
    int voltage_zero_count = 0;
    int current_zero_count = 0;

    memcpy(ADC_DummyBuffer, ADC_Buffer, BUFFER_SIZE * sizeof(uint16_t));

    // Count zero-crossings to detect valid 50 Hz signal
    int voltage_crossings = 0;
    int current_crossings = 0;

    for (int i = 0; i < BUFFER_SIZE / 2; i++) {
        Volt_adcBuffer[i] = ADC_DummyBuffer[2 * i];
        Current_adcBuffer[i] = ADC_DummyBuffer[2 * i + 1];

        float Voltage = ((float)Volt_adcBuffer[i] * 3.3f) / 4095.0f;
        float Current = ((float)Current_adcBuffer[i] * 3.3f) / 4095.0f;

        Voltage_Array[i] = ((Voltage - Voffset) / Gain);
        Current_Array[i] = ((Current - Voffset) / Gain);

        if (Voltage_Array[i] > max_voltage) {
            max_voltage = Voltage_Array[i];
            max_voltage_index = i;
        }
        if (Current_Array[i] > max_current) {
            max_current = Current_Array[i];
            max_current_index = i;
        }

        float abs_Voltage = fabs(Voltage_Array[i]);
        float abs_Current = fabs(Current_Array[i]);
        if (abs_Voltage > max_Actual_Voltage) {
            max_Actual_Voltage = abs_Voltage;
        }
        if (abs_Current > max_Actual_Current) {
            max_Actual_Current = abs_Current;
        }

        // Detect zero-crossings with tolerance
        if (i < BUFFER_SIZE / 2 - 1) {
            if ((Voltage_Array[i] >= -CROSSING_THRESHOLD && Voltage_Array[i + 1] < -CROSSING_THRESHOLD) ||
                (Voltage_Array[i] <= CROSSING_THRESHOLD && Voltage_Array[i + 1] > CROSSING_THRESHOLD)) {
                voltage_crossings++;
            }
            if ((Current_Array[i] >= -CROSSING_THRESHOLD && Current_Array[i + 1] < -CROSSING_THRESHOLD) ||
                (Current_Array[i] <= CROSSING_THRESHOLD && Current_Array[i + 1] > CROSSING_THRESHOLD)) {
                current_crossings++;
            }
        }
    }

    // Calculate effective values
    Effective_Voltage =1.032315978f*115.0f * (2.0 * max_Actual_Voltage);
    Effective_Current = 1.030998694f*7.5 * (2.0 * max_Actual_Current);

    // Debug output to diagnose zero-state issue
    Debug_ADC_Values(max_Actual_Voltage, max_Actual_Current, voltage_crossings, current_crossings);

    // Apply thresholds for zero-state detection
    bool is_zero_state = false;
    // Expect at least 2 zero-crossings for two 50 Hz cycles (1 per cycle)
    if (Effective_Voltage < 30.0f || max_Actual_Voltage < ZERO_THRESHOLD || voltage_crossings < 2) {
        Effective_Voltage = 0.0f;
        is_zero_state = true;
    }
    if (Effective_Current < 1.3f || max_Actual_Current < ZERO_THRESHOLD || current_crossings < 2) { // 1300mA = 1.3A
        Effective_Current = 0.0f;
        is_zero_state = true;
    }

    if (is_zero_state) {
        // Set all derived metrics to zero
        Apparent_Power = 0.0f;
        Real_Power = 0.0f;
        Reactive_Power = 0.0f;
        Phase_Angle = 0.0f;
        Power_factor = 0.0f;
    } else {
        // Proceed with normal calculations
        Apparent_Power = Effective_Voltage * Effective_Current;

        for (int i = 0; i < BUFFER_SIZE / 2 - 1; i++) {
            if (Voltage_Array[i] >= 0 && Voltage_Array[i + 1] < 0) {
                float y1 = Voltage_Array[i];
                float y2 = Voltage_Array[i + 1];
                float x1 = i * 50e-6f;
                float x2 = (i + 1) * 50e-6f;
                float zero_time = x1 - y1 * (x2 - x1) / (y2 - y1);
                voltage_zero_sum += zero_time;
                voltage_zero_count++;
            }
            if (Current_Array[i] >= 0 && Current_Array[i + 1] < 0) {
                float y1 = Current_Array[i];
                float y2 = Current_Array[i + 1];
                float x1 = i * 50e-6f;
                float x2 = (i + 1) * 50e-6f;
                float zero_time = x1 - y1 * (x2 - x1) / (y2 - y1);
                current_zero_sum += zero_time;
                current_zero_count++;
            }
        }

        if (voltage_zero_count > 0 && current_zero_count > 0) {
            voltage_zero_time = voltage_zero_sum / voltage_zero_count;
            current_zero_time = current_zero_sum / current_zero_count;
            float time_diff = current_zero_time - voltage_zero_time;
            Phase_Angle = time_diff * 2.0f * 3.1415926535f * 50.0f;
            while (Phase_Angle > 3.1415926535f) Phase_Angle -= 2.0f * 3.1415926535f;
            while (Phase_Angle < -3.1415926535f) Phase_Angle += 2.0f * 3.1415926535f;
        } else {
            Phase_Angle = 0;
        }

        Real_Power = abs(Effective_Voltage * Effective_Current * cosf(abs(Phase_Angle)));
        Reactive_Power = Effective_Voltage * Effective_Current * sinf(Phase_Angle);
        Power_factor = cosf(Phase_Angle);
    }

    max_Actual_Voltage = 0;
    max_Actual_Current = 0;
}

void middle_button(void) {
    static uint8_t prev_button_state = 1;
    uint32_t current_time = HAL_GetTick();
    uint8_t current_button_state = HAL_GPIO_ReadPin(Middle_Button_GPIO_Port, Middle_Button_Pin);

    if ((prev_button_state == 1) && (current_button_state == 0) && (current_time - last_debounce_time) > BUTTON_DEBOUNCE_TIME_MS) {
        led_state = !led_state;
        HAL_GPIO_WritePin(D2_LED_GPIO_Port, D2_LED_Pin, led_state ? GPIO_PIN_SET : GPIO_PIN_RESET);
        last_debounce_time = current_time;
    }
    prev_button_state = current_button_state;
}
